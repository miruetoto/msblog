박민수 교수 문서의 실자료 여덟 가지를 지점과 기간을 골라 시간 단위로 봄. 바탕색은 그날의 라벨 (사건 · 대조, 빈칸은 라벨 없음). 시각은 현지 표준시. 실은 지점 — 작은 자료 (ASOS · OpenAQ · WEIS · ISD) 는 센 지점 전부, 큰 자료 (에어코리아 · EPA · USGS) 는 시도 · 주마다 사건 · 대조가 가장 고른 지점부터 30곳 (모두 172곳).

idx = FileAttachment("attachments/261015_cfb3d3_raw/index.csv").csv({typed: true})
info = FileAttachment("attachments/261015_7de51b_03.json").json()
COL = ({사건: "#eb6834", 대조: "#2a78d6", 없음: "#b5b3ab", ink: "#0b0b0b", grid: "#e1e0d9", surface: "#fcfcf7", muted: "#898781"})
UNIT = ({에어코리아: "PM2.5 (µg/m³)", OpenAQ: "PM2.5 (µg/m³)", WEIS: "DO (mg/L)", USGS: "DO (mg/L)", "기상청 ASOS ①": "기온 (°C)", "기상청 ASOS ②": "기온 (°C)", "NOAA ISD": "기온 (°C)"})
unitOf = d => d.자료 === "EPA AQS" ? (d.series.endsWith("_o3") ? "O3 (ppm)" : "PM2.5 (µg/m³)") : UNIT[d.자료]
ymd = d => `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, "0")}-${String(d.getUTCDate()).padStart(2, "0")}`
dayIndex = (s, d0) => Math.round((Date.parse(s + "T00:00:00Z") - Date.parse(d0 + "T00:00:00Z")) / 864e5)
LAB = ({"1": "사건", "0": "대조", ".": "없음"})
fd = x => x instanceof Date ? ymd(x) : String(x)
viewof src = Inputs.select([...new Set(idx.map(d => d.자료))], {label: "자료", value: "에어코리아", width: 150})
viewof st = Inputs.select(idx.filter(d => d.자료 === src), {label: "지점", format: d => `${d.지점} · ${d.지역}`, width: 180})
viewof t0 = Inputs.date({label: "시작", value: "2021-01-26", width: 150})
viewof t1 = Inputs.date({label: "끝", value: "2021-02-15", width: 150})
raw = d3.json(`attachments/261015_cfb3d3_raw/${st.file}`)
win = {
  const a = Math.max(0, dayIndex(ymd(t0), raw.d0)), b = Math.min(raw.n - 1, dayIndex(ymd(t1), raw.d0))
  const base = Date.parse(raw.d0 + "T00:00:00Z"), P = raw.P, days = [], pts = []
  for (let i = a; i <= b; i++) {
    const lab = LAB[raw.lab[i]], vals = raw.v.slice(i * P, (i + 1) * P)
    days.push({i, date: new Date(base + i * 864e5), t0: new Date(base + i * 864e5), t1: new Date(base + (i + 1) * 864e5), lab, vals})
    vals.forEach((v, h) => pts.push({t: new Date(base + i * 864e5 + h * 864e5 / P), v, h: h * 24 / P, day: i, lab}))
  }
  return {a, b, days, pts, ok: b >= a}
}
lab1 = info.find(d => d.자료 === st.자료)
md`**${st.지점}** · ${st.지역} · ${st.자료} · ${unitOf(st)} — 자료 ${fd(st.d0)} ~ ${fd(st.d1)} · 쓴 날 사건 ${st.n1} · 대조 ${st.n0}
${win.ok ? `· 고른 기간 ${win.days.length}일 (사건 ${win.days.filter(d => d.lab === "사건").length} · 대조 ${win.days.filter(d => d.lab === "대조").length} · 빈 시각 ${win.pts.filter(d => d.v == null).length})` : "· **고른 기간이 이 지점의 자료 밖입니다**"}

사건 vs 대조 — ${lab1 ? lab1.라벨 : ""}`
win.ok ? Plot.plot({
  width: Math.min(width, 1100), height: 300, marginLeft: 52, style: {background: COL.surface},
  x: {type: "utc", label: null}, y: {label: unitOf(st), grid: true},
  color: {domain: ["사건", "대조"], range: [COL.사건, COL.대조], legend: true},
  marks: [
    Plot.rectX(win.days.filter(d => d.lab !== "없음"), {x1: "t0", x2: "t1", fill: "lab", fillOpacity: 0.16}),
    Plot.lineY(win.pts, {x: "t", y: "v", stroke: COL.ink, strokeWidth: 1.2, tip: {format: {x: d => d.toISOString().slice(0, 16).replace("T", " "), y: true}}})
  ]
}) : html`<p>시작과 끝을 ${fd(st.d0)} ~ ${fd(st.d1)} 안에서 고르십시오.</p>`
curves = win.ok ? win.days.filter(d => d.vals.some(v => v != null)).flatMap(d => d.vals.map((v, h) => ({h: h * 24 / raw.P, v, day: d.i, lab: d.lab}))) : []
means = ["사건", "대조"].flatMap(l => {
  const ds = win.ok ? win.days.filter(d => d.lab === l) : []
  if (!ds.length) return []
  return d3.range(raw.P).map(h => ({h: h * 24 / raw.P, v: d3.mean(ds, d => d.vals[h]), lab: l}))
})
win.ok ? Plot.plot({
  width: Math.min(width, 700), height: 320, marginLeft: 52, style: {background: COL.surface},
  x: {label: "시각 (현지 표준시)", domain: [0, 24], ticks: [0, 6, 12, 18, 24]}, y: {label: unitOf(st), grid: true},
  color: {domain: ["사건", "대조", "없음"], range: [COL.사건, COL.대조, COL.없음], legend: true},
  marks: [
    Plot.lineY(curves, {x: "h", y: "v", z: "day", stroke: "lab", strokeWidth: 0.8, strokeOpacity: 0.45}),
    Plot.lineY(means, {x: "h", y: "v", z: "lab", stroke: "lab", strokeWidth: 3}),
  ]
}) : html``
